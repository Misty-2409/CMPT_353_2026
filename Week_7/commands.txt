-------------------------------------
Create a db table using terminal
-------------------------------------
1. keep docker terminal open
	docker compose up -d
2. open mysql terminal 
	 docker exec -it your-mysql-container-name mysql -u root -p
	for my code 
	 docker exec -it mysql-container mysql -u root -p
	then enter your database password. for my code: admin
3. open your database
	use usersdb; //userdb is database name
4. create table
	 CREATE TABLE posts (
			id INT AUTO_INCREMENT PRIMARY KEY,
			title VARCHAR(255),
			user_id INT,
			FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
	 );
5. insert data
	INSERT INTO posts (title, user_id) VALUES
		('Learning Next.js', 1),
		('Understanding SQL JOIN', 1),
		('Docker Basics', 2);
6. check your database table	
	describe posts;
	select * from posts;
